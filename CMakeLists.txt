cmake_minimum_required(VERSION 3.25)
project(NOESE VERSION 0.1.0 LANGUAGES CXX)
include(CheckCXXCompilerFlag)
check_cxx_compiler_flag("-std=c++26" HAS_CXX26)
if(NOT HAS_CXX26)
  message(FATAL_ERROR "BLOCKED: compiler must accept -std=c++26. No fallback.")
endif()
find_package(SQLite3 REQUIRED)
find_package(OpenSSL REQUIRED)
find_package(Threads REQUIRED)
add_executable(noese src/main.cpp src/Store.cpp src/Service.cpp src/Server.cpp src/Operations.cpp src/Mcp.cpp src/Investigator.cpp)
target_compile_options(noese PRIVATE -std=c++26 -Wall -Wextra -Wpedantic -fmax-errors=3)
target_include_directories(noese PRIVATE src)
target_include_directories(noese SYSTEM PRIVATE third_party)
target_link_libraries(noese PRIVATE SQLite3::SQLite3 OpenSSL::Crypto Threads::Threads)
target_compile_definitions(noese PRIVATE NOESE_ROOT="${CMAKE_CURRENT_SOURCE_DIR}")
enable_testing()
find_package(Python3 REQUIRED COMPONENTS Interpreter)
add_test(NAME vertical COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/vertical.py $<TARGET_FILE:noese>)
set_tests_properties(vertical PROPERTIES TIMEOUT 90)

add_test(NAME mcp COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/mcp.py $<TARGET_FILE:noese>)
set_tests_properties(mcp PROPERTIES TIMEOUT 90)

add_test(NAME investigator COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/investigator.py $<TARGET_FILE:noese>)
set_tests_properties(investigator PROPERTIES TIMEOUT 120)
